   149→    mutationFn: () =>
   150→      apiPost("/api/students", {

   152→        classroomId: classroomId || undefined,

   154→          ...(uniform.top ? ["پیراهن"] : []),

   286→export function StudentProfilePage({ studentId }: { studentId: string }) {
   287→  const { t, locale } = useT();
   288→  const n = useN();
   289→  const qc = useQueryClient();
   290→  const [printKind, setPrintKind] = useState<PrintKind>(null);
   291→  const meta = useQuery({ queryKey: ["meta"], queryFn: () => apiGet<MetaResponse>("/api/meta") });
   292→  const q = useQuery({
   293→    queryKey: ["student", studentId],
   294→    queryFn: () => apiGet<ProfileResponse>(`/api/students/${studentId}`),
   295→  });
   296→  const save = useMutation({
   297→    mutationFn: (data: Record<string, string>) => apiPut(`/api/students/${studentId}`, data),
   298→    onSuccess: () => {
   299→      toast.success(t.saved);
   300→      qc.invalidateQueries({ queryKey: ["student", studentId] });
   301→    },
   302→    onError: (e: Error) => toast.error(e.message),
   303→  });
   304→
   305→  if (q.isLoading || !q.data) return <Loading rows={8} />;
   306→  const { student, enrollments, attendance, invoices, payments, summaries, results, notifications } = q.data;
   307→  const activeEnr = enrollments.find((e) => e.status === "فعال") ?? enrollments[0];
   308→  const schoolInfo: SchoolHeaderInfo | null = meta.data?.school ?? null;
   309→
   310→  const attTotal = attendance.length;
   311→  const attPresent = attendance.filter((a) => a.status === "حاضر").length;
   312→  const attPercent = attTotal > 0 ? Math.round(((attPresent + attendance.filter((a) => a.status === "مریض" || a.status === "رخصتی").length) / attTotal) * 100) : 0;
   313→  const totalBilled = invoices.reduce((a, i) => a + i.amount - i.discount, 0);
   314→  const totalPaid = invoices.reduce((a, i) => a + i.paidAmount, 0);
   315→  const summary = summaries.find((s) => s.enrollmentId === activeEnr?.id);
   316→
   317→  return (
   318→    <div className="space-y-4">
   319→      <div className="flex flex-wrap items-center gap-3">
   320→        <Button variant="outline" size="sm" onClick={() => { window.location.hash = "#/students"; }}>
   321→          <ArrowRight className="h-4 w-4 me-1" /> {t.back}
   322→        </Button>
   323→        <div className="flex-1 min-w-0">
   324→          <h1 className="text-xl font-bold truncate">{student.firstName} {student.fatherName}</h1>
   325→          <p className="text-xs text-muted-foreground" dir="ltr">{student.studentNo}</p>
   326→        </div>
   327→        <div className="flex flex-wrap gap-2">
   328→          <Button size="sm" variant="outline" onClick={() => setPrintKind("tazkira")}><Printer className="h-4 w-4 me-1" /> {t.printTazkiraCard}</Button>
   329→          <Button size="sm" variant="outline" onClick={() => setPrintKind("idcard")}><Printer className="h-4 w-4 me-1" /> {t.printIdCard}</Button>
   330→          <Button size="sm" variant="outline" onClick={() => setPrintKind("regform")}><Printer className="h-4 w-4 me-1" /> {t.printRegForm}</Button>
   331→          <Button size="sm" variant="outline" onClick={() => setPrintKind("results")}><Printer className="h-4 w-4 me-1" /> {t.printResults}</Button>
   332→        </div>
   333→      </div>
   334→
   335→      <Tabs defaultValue="info">
   336→        <TabsList className="flex-wrap h-auto">
   337→          <TabsTrigger value="info">{t.info}</TabsTrigger>
   338→          <TabsTrigger value="attendance">{t.attendanceTab}</TabsTrigger>
   339→          <TabsTrigger value="fees">{t.feesTab}</TabsTrigger>
   340→          <TabsTrigger value="results">{t.resultsTab}</TabsTrigger>
   341→          <TabsTrigger value="docs">{t.docsTab}</TabsTrigger>
   342→        </TabsList>
   343→
   344→        <TabsContent value="info" className="mt-4">
   345→          <InfoTab student={student} onSave={(d) => save.mutate(d)} />
   346→        </TabsContent>
   347→
   348→        <TabsContent value="attendance" className="mt-4">
   349→          <Card>
   350→            <CardHeader className="pb-2">
   351→              <CardTitle className="text-sm">
   352→                {t.attendanceTab}: {n.n(attTotal)} {t.days} — {t.attendancePercent}: {n.n(attPercent)}{t.percentSign}
   353→              </CardTitle>
   354→            </CardHeader>
   355→            <CardContent className="max-h-96 overflow-y-auto nq-scroll">
   356→              {attendance.length === 0 ? <EmptyState text={t.noData} /> : (
   357→                <div className="grid grid-cols-2 sm:grid-cols-4 lg:grid-cols-6 gap-2 text-xs">
   358→                  {attendance.slice(0, 60).map((a) => (
   359→                    <div key={a.id} className="rounded border p-2 flex items-center justify-between gap-1">
   360→                      <JalaliDate iso={a.date} />
   361→                      <StatusBadge status={a.status} />
   362→                    </div>
   363→                  ))}
   364→                </div>
   365→              )}
   366→            </CardContent>
   367→          </Card>
   368→        </TabsContent>
   369→
   370→        <TabsContent value="fees" className="mt-4 space-y-3">
   371→          <div className="grid grid-cols-3 gap-2 text-center text-sm">
   372→            <div className="rounded-lg border p-3"><div className="text-xs text-muted-foreground">{t.total} شقه</div><div className="font-bold">{formatAFN(totalBilled, locale)}</div></div>
   373→            <div className="rounded-lg border p-3"><div className="text-xs text-muted-foreground">{t.payments}</div><div className="font-bold text-emerald-700">{formatAFN(totalPaid, locale)}</div></div>
   374→            <div className="rounded-lg border p-3"><div className="text-xs text-muted-foreground">{t.balance}</div><div className="font-bold text-red-700">{formatAFN(totalBilled - totalPaid, locale)}</div></div>
   375→          </div>
   376→          <DataTable
   377→            rows={invoices as unknown as Array<Record<string, unknown>>}
   378→            searchable={false}
   379→            emptyText={t.noData}
   380→            pageSize={10}
   381→            columns={[
   382→              { key: "monthName", header: t.month, render: (r) => (r as ProfileResponse["invoices"][0]).monthName },
   383→              { key: "amount", header: t.amount, render: (r) => formatAFN((r as ProfileResponse["invoices"][0]).amount) },
   384→              { key: "discount", header: t.discount, render: (r) => formatAFN((r as ProfileResponse["invoices"][0]).discount) },

   390→                  const rem = inv.amount - inv.discount - inv.paidAmount;
   391→                  return <span className={rem > 0 ? "text-red-700 font-semibold" : "text-emerald-700"}>{formatAFN(rem)}</span>;
   392→                },
   393→              },
   394→              {

   401→              },
   402→            ]}
   403→          />
   404→          <h3 className="font-semibold text-sm pt-2">{t.payments}</h3>
   405→          <DataTable
   406→            rows={payments as unknown as Array<Record<string, unknown>>}
   407→            searchable={false}
   408→            emptyText={t.noData}
   409→            pageSize={8}
   410→            columns={[
   411→              { key: "receiptNo", header: t.receiptNo, render: (r) => <span dir="ltr">{(r as ProfileResponse["payments"][0]).receiptNo}</span> },
   412→              { key: "date", header: t.date, render: (r) => <JalaliDate iso={(r as ProfileResponse["payments"][0]).date} /> },
   413→              { key: "amount", header: t.amount, render: (r) => formatAFN((r as ProfileResponse["payments"][0]).amount) },

   415→            ]}

   418→
   419→        <TabsContent value="results" className="mt-4">
   420→          <Card>
   421→            <CardHeader className="pb-2">
   422→              <CardTitle className="text-sm">
   423→                {t.resultsTab} {summary ? `— ${t.average}: ${n.n(summary.average)} | ${t.rank}: ${n.n(summary.rank)} | ` : ""}
